199  // Simple sentiment word lists

200  const positiveWords = ['help', 'good', 'better', 'vaccine', 'relief', 'recovery'];

201  const negativeWords = ['terrible', 'worse', 'worried', 'severe', 'pain', 'panic', 'sick', 'chills', 'fever', 'cough', 'headache'];

203  // Utility: Simple sentiment analyzer for a text snippet

204  function analyzeSentiment(text) {

205    const txt = text.toLowerCase();

206    let score = 0;

207    positiveWords.forEach(word => { if (txt.includes(word)) score++; });

208    negativeWords.forEach(word => { if (txt.includes(word)) score--; });

209    if(score > 0) return 'Positive';

210    if(score < 0) return 'Negative';

211    return 'Neutral';

212  }

214  // Data cleaning: Remove duplicates, advertisements (simulate by filtering out certain keywords)

215  function cleanData(posts) {

216    const filtered = [];

217    const seenTexts = new Set();

218    const adKeywords = ['buy now', 'sale', 'offer', 'discount', 'advertisement'];

219

220    for(const p of posts) {

221      const txtLower = p.text.toLowerCase();

222      if(adKeywords.some(kw => txtLower.includes(kw))) continue;

223      if(seenTexts.has(txtLower)) continue;

224      seenTexts.add(txtLower);

225      filtered.push(p);

226    }

227    return filtered;

228  }

261  // Anomaly detection: identifying dates with spikes over mean + 1.5*std

262  function detectAnomalies(counts) {

263    const values = Object.values(counts);

264    const mean = values.reduce((a,b) => a+b,0) / values.length;

265    const variance = values.reduce((a,b) => a + Math.pow(b - mean, 2), 0) / values.length;

266    const stdDev = Math.sqrt(variance);

267

268    const threshold = mean + 1.5 * stdDev;

269    const anomalies = [];

270    for(const [date, val] of Object.entries(counts)) {

271      if(val > threshold) {

272        anomalies.push({date, count: val});

273      }

274    }

275    return {

276      mean,

277      stdDev,

278      threshold,

279      anomalies

280    };

281  }
